Q2.  Write a program that implements a multi-thread application that has three threads. First 
thread generates random integer after every 1 second and if the value is even, second 
thread computes the square of the number and prints. If the value is odd, the third 
thread will print the value of cube of the number.

  public class MultiThreadDemo {
    static int n;
    static boolean isNew = false;
    static final Object lock = new Object();

    // Helper method to create Thread 2 (Even) and Thread 3 (Odd)
    static void startWorker(boolean isEven) {
        new Thread(() -> {
            try {
                while (true) {
                    synchronized (lock) {
                        // Wait until there is a new number matching this thread's condition
                        while (!isNew || (n % 2 == 0) != isEven) lock.wait();
                        
                        System.out.println(isEven ? "Square: " + (n * n) : "Cube: " + (n * n * n));
                        isNew = false; // Reset flag after processing
                    }
                }
            } catch (Exception e) {}
        }).start();
    }

    public static void main(String[] args) {
        startWorker(true);  // Thread 2: Handles Even numbers
        startWorker(false); // Thread 3: Handles Odd numbers

        // Thread 1: Generates random numbers every 1 second
        new Thread(() -> { 
            try {
                while (true) {
                    synchronized (lock) {
                        n = (int) (Math.random() * 100);
                        System.out.println("\nGenerated: " + n);
                        isNew = true;
                        lock.notifyAll(); // Wake up waiting threads
                    }
                    Thread.sleep(1000); 
                }
            } catch (Exception e) {}
        }).start();
    }
}

  Q2. Write a SERVLET program which counts how many times a user has visited a web 
page. If user is visiting the page for the first time, display a welcome message. If the 
user is revisiting the page, display the number of times visited. (Use Cookie) 

  import java.io.IOException;
import javax.servlet.annotation.WebServlet;
import javax.servlet.http.*;

@WebServlet("/VisitServlet")
public class VisitServlet extends HttpServlet {
    protected void doGet(HttpServletRequest req, HttpServletResponse res) throws IOException {
        res.setContentType("text/html");
        int count = 1;
        Cookie[] cookies = req.getCookies();
        
        // Check for existing cookie
        if (cookies != null) {
            for (Cookie c : cookies) {
                if ("visitCount".equals(c.getName())) {
                    count = Integer.parseInt(c.getValue()) + 1;
                }
            }
        }
        
        // Update cookie with new count
        res.addCookie(new Cookie("visitCount", String.valueOf(count)));
        
        // Display output
        res.getWriter().print("<h3>" + (count == 1 ? "Welcome! This is your first visit." : "You have visited " + count + " times.") + "</h3>");
    }
}

import java.util.*;

public class ColorList {
    public static void main(String[] args) {
        LinkedList<String> list = new LinkedList<>(Arrays.asList("red", "blue", "yellow", "orange"));
        Scanner sc = new Scanner(System.in);
        
        for (;;) {
            System.out.print("\n1.Display 2.Reverse 3.Insert 4.Exit\nChoice: ");
            int ch = sc.nextInt();
            
            if (ch == 1) {
                // a. Display using Iterator
                Iterator<String> it = list.iterator();
                while (it.hasNext()) System.out.print(it.next() + " ");
            } 
            else if (ch == 2) {
                // b. Display in reverse using ListIterator
                ListIterator<String> lit = list.listIterator(list.size());
                while (lit.hasPrevious()) System.out.print(lit.previous() + " ");
            } 
            else if (ch == 3) {
                // c. Insert pink and green between blue and yellow
                list.addAll(list.indexOf("blue") + 1, Arrays.asList("pink", "green"));
                System.out.println("Updated List: " + list);
            } 
            else break;
        }
        sc.close();
    }
}

import javax.swing.*;
import java.awt.FlowLayout;

public class AlphabetThread {
    public static void main(String[] args) {
        JFrame f = new JFrame("A-Z Continuous");
        JTextField tf = new JTextField(5);
        JButton b = new JButton("Start");

        b.addActionListener(e -> new Thread(() -> {
            while (true) { // Loops continuously
                for (char c = 'A'; c <= 'Z'; c++) {
                    tf.setText(String.valueOf(c));
                    try { Thread.sleep(200); } catch (Exception ex) {} // 200ms delay
                }

        }).start());

        f.add(tf);
        f.add(b);
        f.setSize(250, 100);
        f.setDefaultCloseOperation(JFrame.EXIT_ON_CLOSE);
        f.setVisible(true);
    }
}

Q2. Write a java program to solve producer consumer problem in which a producer 
produces a value and consumer consume the value before producer generate the next 
value. (Use thread synchronization) 

public class ProducerConsumer {
    static int val;
    static boolean hasVal = false;
    static final Object lock = new Object();

    public static void main(String[] args) {
        // Producer Thread
        new Thread(() -> { 
            try {
                for (int i = 1; i <= 5; i++) {
                    synchronized (lock) {
                        while (hasVal) lock.wait(); // Wait if buffer is full
                        System.out.println("Produced: " + (val = i));
                        hasVal = true;
                        lock.notify(); // Wake up consumer
                    }
                }
            } catch (Exception e) {}
        }).start();

        // Consumer Thread
        new Thread(() -> { 
            try {
                for (int i = 1; i <= 5; i++) {
                    synchronized (lock) {
                        while (!hasVal) lock.wait(); // Wait if buffer is empty
                        System.out.println("Consumed: " + val);
                        hasVal = false;
                        lock.notify(); // Wake up producer
                    }
                }
            } catch (Exception e) {}
        }).start();
    }
}

import javax.servlet.annotation.WebServlet;
import javax.servlet.http.*;
import java.io.IOException;

@WebServlet("/visit")
public class VisitCounter extends HttpServlet {
    public void doGet(HttpServletRequest req, HttpServletResponse res) throws IOException {
        int count = 1;
        if (req.getCookies() != null) 
            for (Cookie c : req.getCookies()) 
                if (c.getName().equals("v")) count = Integer.parseInt(c.getValue()) + 1;
        
        res.addCookie(new Cookie("v", count + ""));
        res.getWriter().print("Total Visits: " + count);
    }
}
